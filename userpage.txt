    <script>
        const pfppic = document.getElementById('pfpPic'); // image input
        const pfppicoutput = document.getElementById('pfp-pic-img'); // image output
        const pfppicname = document.querySelector('span.pfpPic-name');
        const editpfppic = document.getElementById('edit-pfpPic'); // image input
        const editpfppicoutput = document.getElementById('edit-pfp-pic-img'); // image output
        const editpfppicname = document.querySelector('span.edit-pfpPic-name');
        //add user
        const btnOpen = document.querySelector('.btn-add-new-user');
        const btnClose = document.getElementById('btn-add-new-user-close');
        const formContainer = document.querySelector('.form-content-container');
        //add user -- password show/hide toggle
        const PWtype = document.querySelector('.form-content-input-field-pw');
        const PWinput = document.getElementById('password');
        //edit user -- password show/hide toggle
        const editUserPWtype = document.querySelector('.edit-user-content-input-field-pw');
        const editUserPWinput = document.getElementById('edit-password');
        
        //view profile
        const userProfileOpen = document.querySelector('.user-account-profile');
        const userProfileClose = document.querySelector('.btn-view-user-close');
        const viewUserAcc = document.querySelector('.users-account-content-data-view-user-content');
        //edit profile
        const userEditOpen = document.querySelector('.user-account-edit');
        const userEditClose = document.querySelector('.btn-edit-user-close');
        const editUserAcc = document.querySelector('.users-account-content-data-edit-user-content');
        

        btnOpen.addEventListener('click', function () {
            formContainer.style.display = 'flex';
        });
        btnClose.addEventListener('click', function () {
            formContainer.style.display = 'none';
        });
        userProfileOpen.addEventListener('click', function () {
            viewUserAcc.style.display = 'flex';
        });
        userProfileClose.addEventListener('click', function () {
            viewUserAcc.style.display = 'none';
        });
        userEditOpen.addEventListener('click', function () {
            editUserAcc.style.display = 'flex';
        });
        userEditClose.addEventListener('click', function () {
            editUserAcc.style.display = 'none';
        });

        pfppic.addEventListener('change', function() {
            pfppicoutput.style.display = 'flex';
            const file = pfppic.files[0]; // Get the selected file
            const fileName = file.name; // Extract the file name
            const fileReader = new FileReader();

            fileReader.onload = function() {
                pfppicoutput.src = fileReader.result; // Display the image
            };

            fileReader.readAsDataURL(file);
            pfppicname.innerHTML = fileName;
        });
        editpfppic.addEventListener('change', function() {
            editpfppicoutput.style.display = 'flex';
            const file = editpfppic.files[0]; // Get the selected file
            const fileName = file.name; // Extract the file name
            const fileReader = new FileReader();

            fileReader.onload = function() {
                editpfppicoutput.src = fileReader.result; // Display the image
                editpfppicname.innerHTML = fileName;
            };

            fileReader.readAsDataURL(file);
        });
        
        PWtype.addEventListener('click', function () {
            if (PWinput.type === 'password') {
                PWtype.classList.remove('password-hide');
                PWtype.classList.add('password-show');
                PWinput.type = 'text';
            } else {
                PWtype.classList.add('password-hide');
                PWtype.classList.remove('password-show');
                PWinput.type = 'password';
            }
        });
        editUserPWtype.addEventListener('click', function () {
            if (editUserPWinput.type === 'password') {
                editUserPWtype.classList.remove('edit-password-hide');
                editUserPWtype.classList.add('edit-password-show');
                editUserPWinput.type = 'text';
            } else {
                editUserPWtype.classList.add('edit-password-hide');
                editUserPWtype.classList.remove('edit-password-show');
                editUserPWinput.type = 'password';
            }
        });
    </script>
